nf-metalog groups task events by a metadata key (e.g. meta.id), persists them to SQLite (optional) and generates interactive HTML reports so you can follow every sample end-to-end.

Quick start

Add this to your nextflow.config and you're done.

plugins {
    id "nf-metalog@0.1.0"
}

metalog {
    enabled        = true
    groupKey       = 'id'          // key in your meta map
    storageBackend = 'sqlite'     // or 'memory'
    report {
        csvFile  = 'metalog.csv'
        htmlFile = 'metalog.html'
    }
}
